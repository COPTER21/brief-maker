// ---- 2. State (one per combobox instance) -----------------------------------

const empSearchState = { query: '', open: false, highlight: 0 };

// ---- 3. Field markup (paste inside your form template string) ---------------

// ---- 4. Handlers (verbatim) -------------------------------------------------

function renderEmpSuggestions() {
      const q = (empSearchState.query || '').toLowerCase().trim();
      const filtered = mockEmployees.filter(e => !q || e.name.toLowerCase().includes(q) || e.position.toLowerCase().includes(q) || e.dept.toLowerCase().includes(q));
      if (filtered.length === 0) return `<div style="padding: 24px 16px; text-align: center; font-size: 13px; color: var(--c-text-mute);">ไม่พบพนักงานที่ตรงกับ "<strong>${q}</strong>"</div>`;
      return filtered.map((e, i) => {
        const active = i === empSearchState.highlight;
        return `
          <button type="button" onmousedown="event.preventDefault()" onclick="selectEmp('${e.id}')"
            style="width: 100%; text-align: left; padding: 9px 12px; background: ${active ? 'var(--c-primary-08)' : 'transparent'}; border: none; cursor: pointer; display: flex; align-items: center; gap: 11px; border-bottom: 1px solid var(--c-border-soft);">
            <div style="width: 34px; height: 34px; border-radius: 50%; background: linear-gradient(135deg, var(--c-primary), var(--c-teal)); color: #fff; display: grid; place-items: center; font-size: 12px; font-weight: 600; flex-shrink: 0;">${e.initials}</div>
            <div style="flex: 1; min-width: 0;">
              <div style="font-size: 13px; font-weight: 600; color: var(--c-navy);">${e.name}</div>
              <div style="font-size: 11.5px; color: var(--c-text-mute); white-space: nowrap; overflow: hidden; text-overflow: ellipsis;">${e.position} · ${e.dept}</div>
            </div>
          </button>`;
      }).join('');
    }

function onEmpSearchInput(e) {
      empSearchState.query = e.target.value; empSearchState.open = true; empSearchState.highlight = 0;
      if (createWizard.data.requester && !mockEmployees.find(x => x.name === e.target.value)) createWizard.data.requester = '';
      const list = document.getElementById('empSuggestList');
      if (list) { list.classList.remove('hidden'); list.innerHTML = renderEmpSuggestions(); renderIcons(); }
    }

function onEmpSearchFocus() { empSearchState.open = true; const l = document.getElementById('empSuggestList'); if (l) { l.classList.remove('hidden'); l.innerHTML = renderEmpSuggestions(); renderIcons(); } }

function onEmpSearchKey(e) {
      const list = document.getElementById('empSuggestList'); if (!list) return;
      const q = (empSearchState.query || '').toLowerCase().trim();
      const filtered = mockEmployees.filter(x => !q || x.name.toLowerCase().includes(q) || x.position.toLowerCase().includes(q) || x.dept.toLowerCase().includes(q));
      if (e.key === 'ArrowDown') { e.preventDefault(); empSearchState.highlight = Math.min(filtered.length - 1, empSearchState.highlight + 1); list.innerHTML = renderEmpSuggestions(); renderIcons(); }
      else if (e.key === 'ArrowUp') { e.preventDefault(); empSearchState.highlight = Math.max(0, empSearchState.highlight - 1); list.innerHTML = renderEmpSuggestions(); renderIcons(); }
      else if (e.key === 'Enter') { e.preventDefault(); if (filtered[empSearchState.highlight]) selectEmp(filtered[empSearchState.highlight].id); }
      else if (e.key === 'Escape') { e.stopPropagation(); empSearchState.open = false; list.classList.add('hidden'); }
    }

function selectEmp(id) {
      const emp = mockEmployees.find(e => e.id === id);
      if (emp) createWizard.data.requester = emp.name;   // เก็บแค่ชื่อ
      empSearchState.open = false; empSearchState.query = '';
      renderCreateDrawer();
    }

function clearEmpSelection() { createWizard.data.requester = ''; empSearchState.query = ''; empSearchState.open = true; renderCreateDrawer(); setTimeout(() => document.getElementById('empSearchInput')?.focus(), 0); }

// ---- 5. Outside-click close (register ONCE per page, list every combobox) ---
document.addEventListener('mousedown', (e) => {
  [['empCombobox','empSuggestList',empSearchState]].forEach(([boxId, listId, st]) => {
    const box = document.getElementById(boxId);
    if (box && !box.contains(e.target)) { st.open = false; document.getElementById(listId)?.classList.add('hidden'); }
  });
});
