    // Migrate legacy lines (vat_enabled/net_enabled) to vat_mode
    function migrateLine(l) {
      if (l.vat_mode) return l;  // already migrated
      let mode = 'none';
      if (l.vat_enabled && l.net_enabled) mode = 'included';
      else if (l.vat_enabled) mode = 'add';
      l.vat_mode = mode;
      return l;
    }

    // Calculate VAT for a line based on vat_mode
    // Returns { vatAmount, lineTotal }
    //   none      → no VAT, line = subtotal after discount
    //   add       → VAT added on top, line = subtotal × (1 + vat%)
    //   included  → VAT included in price, line = subtotal; VAT base = subtotal / (1 + vat%)
    function calcLineVat(l) {
      migrateLine(l);
      const subtotal = l.qty * l.unit_price;
      const discount = subtotal * (l.discount_pct / 100);
      const netAmount = subtotal - discount;
      let vatAmount = 0;
      let lineTotal = netAmount;
      if (l.vat_mode === 'add') {
        vatAmount = netAmount * (l.vat_pct / 100);
        lineTotal = netAmount + vatAmount;
      } else if (l.vat_mode === 'included') {
        vatAmount = netAmount * (l.vat_pct / 100) / (1 + l.vat_pct / 100);
        lineTotal = netAmount;  // line total stays same; VAT is back-calculated
      }
      return { netAmount, vatAmount, lineTotal };
    }

    function escapeReg(s) { return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }

    function onBrSearchInput(e) {
      brSearchState.query = e.target.value;
      brSearchState.open = true; brSearchState.highlight = 0;
      if (createWizard.data.br_no) createWizard.data.br_no = null;
      const list = document.getElementById('brSuggestList');
      if (list) { list.classList.remove('hidden'); list.innerHTML = renderBrSuggestions(); renderIcons(); }
    }
    function onBrSearchFocus() {
      brSearchState.open = true;
      const list = document.getElementById('brSuggestList');
      if (list) { list.classList.remove('hidden'); list.innerHTML = renderBrSuggestions(); renderIcons(); }
    }
    function onBrSearchKey(e) {
      const list = document.getElementById('brSuggestList');
      if (!list) return;
      const filtered = mockBudgets.filter(b => {
        const q = (brSearchState.query || '').toLowerCase().trim();
        return !q || b.br_no.toLowerCase().includes(q) || b.name.toLowerCase().includes(q);
      });
      if (e.key === 'ArrowDown') { e.preventDefault(); brSearchState.highlight = Math.min(filtered.length - 1, brSearchState.highlight + 1); list.innerHTML = renderBrSuggestions(); renderIcons(); }
      else if (e.key === 'ArrowUp') { e.preventDefault(); brSearchState.highlight = Math.max(0, brSearchState.highlight - 1); list.innerHTML = renderBrSuggestions(); renderIcons(); }
      else if (e.key === 'Enter') { e.preventDefault(); if (filtered[brSearchState.highlight]) selectBr(filtered[brSearchState.highlight].br_no); }
      else if (e.key === 'Escape') { brSearchState.open = false; list.classList.add('hidden'); }
    }

      brSearchState.open = false; brSearchState.query = '';

      const list = document.getElementById('ucpSuggestList');
      if (!list) return;
      const filtered = mockUCPs.filter(u => {
        const q = (ucpSearchState.query || '').toLowerCase().trim();
        return !q || u.ucp_no.toLowerCase().includes(q) || u.desc.toLowerCase().includes(q);
      });
      if (e.key === 'ArrowDown') { e.preventDefault(); ucpSearchState.highlight = Math.min(filtered.length - 1, ucpSearchState.highlight + 1); list.innerHTML = renderUcpSuggestions(); renderIcons(); }
      else if (e.key === 'ArrowUp') { e.preventDefault(); ucpSearchState.highlight = Math.max(0, ucpSearchState.highlight - 1); list.innerHTML = renderUcpSuggestions(); renderIcons(); }
      else if (e.key === 'Enter') { e.preventDefault(); if (filtered[ucpSearchState.highlight]) selectUcp(filtered[ucpSearchState.highlight].ucp_no); }
      else if (e.key === 'Escape') { ucpSearchState.open = false; list.classList.add('hidden'); }
    }
    function selectUcp(ucp_no) {
      createWizard.data.ucp_no = ucp_no;
      ucpSearchState.open = false; ucpSearchState.query = '';
      renderCreateDrawer();
    }

      setTimeout(() => document.getElementById('ucpSearchInput')?.focus(), 0);
    }

    function toggleLineExpand(lineId) {
      const wasOpen = createWizard.expandedLineId === lineId;

      if (wasOpen) {
        // Collapse: find the existing panel, apply collapse animation, then remove after animation
        const existingPanel = document.querySelector(`.line-expand-panel-anchor[data-line-id="${lineId}"] .line-expand-panel-inner`);
        if (existingPanel) {
          existingPanel.classList.add('line-collapse-panel');
          setTimeout(() => {
            createWizard.expandedLineId = null;
            createWizard._justOpenedLineId = null;
            renderCreateDrawer();
          }, 200);
          return;
        }
        // Fallback if panel not found
        createWizard.expandedLineId = null;
        createWizard._justOpenedLineId = null;
        renderCreateDrawer();
      } else {
        // Open: if another line was open, collapse it first (instantly), then open this one with animation
        createWizard.expandedLineId = lineId;
        createWizard._justOpenedLineId = lineId;
        renderCreateDrawer();
        createWizard._justOpenedLineId = null;
      }
    }

    function removeLine(id) {
      const idx = createWizard.data.lines.findIndex(l => l.id === id);
      if (idx >= 0) createWizard.data.lines.splice(idx, 1);
      if (createWizard.data.lines.length === 0) createWizard.data.lines.push(createBlankLine());
      renderCreateDrawer();
    }
    function updateLine(id, field, val) {
      const line = createWizard.data.lines.find(l => l.id === id);
      if (!line) return;
      // Save focus
      const focused = document.activeElement;
      const focusInfo = focused && focused.tagName === 'INPUT' ? {
        oninput: focused.getAttribute('oninput'),
        selStart: focused.selectionStart,
      } : null;

      if (field === 'qty' || field === 'unit_price' || field === 'discount_pct' || field === 'vat_pct') {
        line[field] = parseFloat(val) || 0;
      } else if (field === 'vat_mode') {
        // Mutually exclusive: 'none' | 'add' | 'included'
        line.vat_mode = val;
        // Cleanup legacy fields if present
        delete line.net_enabled;
        delete line.vat_enabled;
      } else {
        line[field] = val;
      }
      renderCreateDrawer();

      // Restore focus
      if (focusInfo) {
        const inputs = document.querySelectorAll('input');
        for (const inp of inputs) {
          if (inp.getAttribute('oninput') === focusInfo.oninput) {
            inp.focus();
            try { inp.setSelectionRange(focusInfo.selStart, focusInfo.selStart); } catch(e) {}
            break;
          }
        }
      }
    }

      // Simple click-to-fill popover — built fresh per call

      pop.style.cssText = `position: fixed; left: ${rect.left}px; top: ${rect.bottom + 2}px; min-width: 320px; background: white; border: 1px solid var(--c-border); border-radius: var(--r); box-shadow: var(--sh-lg); z-index: 70; max-height: 240px; overflow-y: auto;`;

    function wizardNext() {
      const s = createWizard.step;
      if (s === 1) {
        if (!createWizard.data.pr_date) return showToast('กรุณาระบุวันที่ขอซื้อ', 'warning');
        if (!createWizard.data.need_date) return showToast('กรุณาระบุวันที่ต้องการสินค้า', 'warning');
      }
      if (s === 2) {
        const validLines = createWizard.data.lines.filter(l => l.item_code && l.qty > 0 && l.unit_price > 0);
        if (validLines.length === 0) return showToast('กรุณาเพิ่มรายการสินค้าอย่างน้อย 1 รายการ', 'warning');
      }
      createWizard.step++;
      renderCreateDrawer();
    }
    function wizardBack() {
      createWizard.step = Math.max(1, createWizard.step - 1);
      renderCreateDrawer();
    }
    function saveDraft() {
      const next = createNewRecord('draft');
      state.records.unshift(next);
      showToast(`บันทึกแบบร่าง ${next.pr_no} เรียบร้อย`, 'success');
      closeCreateDrawer();
    }
    function submitForApproval() {
      const next = createNewRecord('pending');
      state.records.unshift(next);
      showToast(`ส่ง ${next.pr_no} เพื่อขออนุมัติแล้ว`, 'success');
      closeCreateDrawer();
    }
    function createNewRecord(status) {
      const d = createWizard.data;
      const today = new Date();
      const buddhistShort = (today.getFullYear() + 543).toString().slice(-2);
      const seq = String(state.records.length + 1).padStart(5, '0');
      let priceBeforeVat = 0, totalVat = 0;
      d.lines.forEach(l => {
        migrateLine(l);
        const { netAmount, vatAmount } = calcLineVat(l);
        if (l.vat_mode === 'included') priceBeforeVat += netAmount - vatAmount;
        else priceBeforeVat += netAmount;
        totalVat += vatAmount;
      });
      const subtotalAfterVat = priceBeforeVat + totalVat;
      const whtAmount = d.wht_enabled ? priceBeforeVat * ((d.wht_pct || 0) / 100) : 0;
      let endbillAmount = 0;
      if (d.endbill_discount_enabled) {
        if (d.endbill_discount_mode === 'amount') endbillAmount = d.endbill_discount_value || 0;
        else endbillAmount = subtotalAfterVat * ((d.endbill_discount_value || 0) / 100);
      }
      const grandTotal = subtotalAfterVat - whtAmount - endbillAmount;
      return {
        id: 'new-' + Date.now(),
        pr_no: `PR-${buddhistShort}${String(today.getMonth() + 1).padStart(2, '0')}-${seq}`,
        pr_date: d.pr_date,
        need_date: d.need_date,
        requester: d.requester,
        dept: d.dept,
        type: 'สินค้า',
        br_no: d.br_no,
        br_code: d.br_no ? mockBudgets.find(b => b.br_no === d.br_no)?.code : null,
        ucp_no: d.ucp_no,
        status,
        order_status: 'NONE',
        ordered_pct: 0,
        grand_total: grandTotal,
        updated_at: today.toISOString().slice(0, 10),
      };
    }
