function createBlankLine() {
      return {
        id: 'L' + Date.now() + Math.random().toString(36).slice(2, 5),
        item_code: '',
        item_name: '',
        qty: 1,
        unit: '-',
        unit_price: 0,
        discount_pct: 0,
        vat_mode: 'none',     // 'none' | 'add' | 'included' — mutually exclusive
        vat_pct: 7,
        note: '',
      };
    }

function migrateLine(l) {
      if (l.vat_mode) return l;  // already migrated
      let mode = 'none';
      if (l.vat_enabled && l.net_enabled) mode = 'included';
      else if (l.vat_enabled) mode = 'add';
      l.vat_mode = mode;
      return l;
    }

function calcLineVat(l) {
      migrateLine(l);
      const subtotal = l.qty * l.unit_price;
      const discount = subtotal * (l.discount_pct / 100);
      const netAmount = subtotal - discount;
      let vatAmount = 0;
      let lineTotal = netAmount;
      if (l.vat_mode === 'add') {
        vatAmount = netAmount * (l.vat_pct / 100);
        lineTotal = netAmount + vatAmount;
      } else if (l.vat_mode === 'included') {
        vatAmount = netAmount * (l.vat_pct / 100) / (1 + l.vat_pct / 100);
        lineTotal = netAmount;  // line total stays same; VAT is back-calculated
      }
      return { netAmount, vatAmount, lineTotal };
    }

function toggleLineExpand(lineId) {
      const wasOpen = createWizard.expandedLineId === lineId;

      if (wasOpen) {
        // Collapse: find the existing panel, apply collapse animation, then remove after animation
        const existingPanel = document.querySelector(`.line-expand-panel-anchor[data-line-id="${lineId}"] .line-expand-panel-inner`);
        if (existingPanel) {
          existingPanel.classList.add('line-collapse-panel');
          setTimeout(() => {
            createWizard.expandedLineId = null;
            createWizard._justOpenedLineId = null;
            renderCreateDrawer();
          }, 200);
          return;
        }
        // Fallback if panel not found
        createWizard.expandedLineId = null;
        createWizard._justOpenedLineId = null;
        renderCreateDrawer();
      } else {
        // Open: if another line was open, collapse it first (instantly), then open this one with animation
        createWizard.expandedLineId = lineId;
        createWizard._justOpenedLineId = lineId;
        renderCreateDrawer();
        createWizard._justOpenedLineId = null;
      }
    }

function updateLineSummaryOnly() {
      const wrap = document.getElementById('line-summary-wrap');
      if (wrap) wrap.innerHTML = renderLineSummary();
    }

function renderLineSummary() {
      const lines = createWizard.data.lines;
      const wht = createWizard.data.wht_enabled;
      const whtPct = createWizard.data.wht_pct || 0;
      const endbillEnabled = createWizard.data.endbill_discount_enabled;
      const endbillMode = createWizard.data.endbill_discount_mode;
      const endbillValue = createWizard.data.endbill_discount_value || 0;

      let priceBeforeVat = 0;
      let totalVat = 0;
      lines.forEach(l => {
        migrateLine(l);
        const { netAmount, vatAmount } = calcLineVat(l);
        // priceBeforeVat = revenue net of VAT
        //   'none'     → no VAT, base = netAmount
        //   'add'      → price excludes VAT, base = netAmount
        //   'included' → price includes VAT, base = netAmount - vatAmount
        if (l.vat_mode === 'included') {
          priceBeforeVat += netAmount - vatAmount;
        } else {
          priceBeforeVat += netAmount;
        }
        totalVat += vatAmount;
      });

      const subtotalAfterVat = priceBeforeVat + totalVat;
      const whtAmount = wht ? priceBeforeVat * (whtPct / 100) : 0;
      let endbillAmount = 0;
      if (endbillEnabled) {
        if (endbillMode === 'amount') endbillAmount = endbillValue;
        else endbillAmount = subtotalAfterVat * (endbillValue / 100);
      }
      const grandTotal = subtotalAfterVat - whtAmount - endbillAmount;

      // Hard Control — PR (สะสมทุกใบบน BR เดียวกัน) ห้ามเกินงบคงเหลือของใบปลดอายัด
      const selBr = createWizard.data.br_no ? mockBudgets.find(b => b.br_no === createWizard.data.br_no) : null;
      const overBudget = !!(selBr && grandTotal > selBr.remaining + 0.0001);
      createWizard._overBudget = overBudget;
      const overWarn = overBudget ? `
        <div style="display: flex; align-items: flex-start; gap: 8px; padding: 10px 12px; margin-bottom: 10px; border-radius: var(--r); background: rgba(239,68,68,0.08); border: 1px solid rgba(239,68,68,0.25);">
          <i data-lucide="alert-triangle" class="w-4 h-4" style="width:16px;height:16px;color:var(--c-danger);flex-shrink:0;margin-top:1px;"></i>
          <div style="font-size: 12px; color: var(--c-danger); line-height: 1.5;">
            <strong>ยอด PR เกินงบคงเหลือของใบปลดอายัด</strong> — งบคงเหลือ ฿${formatMoney(selBr.remaining)} · ยอดนี้ ฿${formatMoney(grandTotal)}.
            ลดยอด หรือเลือกใบปลดอายัดอื่น (Hard Control — ส่งอนุมัติไม่ได้)
          </div>
        </div>` : '';

      return `
        ${overWarn}
        <div class="card">
          <div class="card-head" style="padding: 12px 18px;">
            <div style="display: flex; align-items: center; gap: 8px;">
              <i data-lucide="calculator" class="w-4 h-4" style="color: var(--c-primary);"></i>
              <div class="card-title">สรุปยอด (${lines.length} รายการ)</div>
            </div>
            <span class="pill" style="background: rgba(124,58,237,0.10); color: var(--c-purple); padding: 1px 7px; font-size: 10px; letter-spacing: 0.04em;">BY_LINE</span>
          </div>
          <div class="card-body" style="padding: 12px 18px;">
            <table style="width: 100%; font-size: 13px;">
              <tbody>
                <tr><td style="padding: 4px 0;">ราคาก่อน VAT</td><td style="text-align: right; font-variant-numeric: tabular-nums; padding: 4px 0; font-weight: 600;">฿${formatMoney(priceBeforeVat)}</td></tr>
                <tr><td style="padding: 4px 0; color: var(--c-purple);">ภาษีมูลค่าเพิ่ม (VAT)</td><td style="text-align: right; font-variant-numeric: tabular-nums; padding: 4px 0; color: var(--c-purple);">+฿${formatMoney(totalVat)}</td></tr>
                <tr style="border-top: 1px solid var(--c-border-soft);"><td style="padding: 6px 0;">ยอดรวมหลัง VAT</td><td style="text-align: right; font-variant-numeric: tabular-nums; padding: 6px 0; font-weight: 600;">฿${formatMoney(subtotalAfterVat)}</td></tr>
                ${wht ? `<tr><td style="padding: 4px 0; color: var(--c-warning);">หัก ภาษี ณ ที่จ่าย ${whtPct}%</td><td style="text-align: right; font-variant-numeric: tabular-nums; padding: 4px 0; color: var(--c-warning);">-฿${formatMoney(whtAmount)}</td></tr>` : ''}
                ${endbillEnabled ? `<tr><td style="padding: 4px 0; color: var(--c-text-mute);">ส่วนลดท้ายบิล${endbillMode === 'percent' ? ' (' + endbillValue + '%)' : ''}</td><td style="text-align: right; font-variant-numeric: tabular-nums; padding: 4px 0; color: var(--c-text-mute);">-฿${formatMoney(endbillAmount)}</td></tr>` : ''}
                <tr style="border-top: 2px solid var(--c-border);"><td style="padding: 8px 0 0; font-weight: 700; font-size: 14px;">ยอดสุทธิทั้งหมด</td><td style="text-align: right; font-variant-numeric: tabular-nums; padding: 8px 0 0; font-weight: 700; font-size: 16px; color: var(--c-primary);">฿${formatMoney(grandTotal)}</td></tr>

            </table>
          </div>

    }

function addLine() {
      createWizard.data.lines.push(createBlankLine());
      renderCreateDrawer();
    }

function removeLine(id) {
      const idx = createWizard.data.lines.findIndex(l => l.id === id);
      if (idx >= 0) createWizard.data.lines.splice(idx, 1);
      if (createWizard.data.lines.length === 0) createWizard.data.lines.push(createBlankLine());
      renderCreateDrawer();
    }

function updateLine(id, field, val) {
      const line = createWizard.data.lines.find(l => l.id === id);
      if (!line) return;
      // Save focus
      const focused = document.activeElement;
      const focusInfo = focused && focused.tagName === 'INPUT' ? {
        oninput: focused.getAttribute('oninput'),
        selStart: focused.selectionStart,
      } : null;

      if (field === 'qty' || field === 'unit_price' || field === 'discount_pct' || field === 'vat_pct') {
        line[field] = parseFloat(val) || 0;
      } else if (field === 'vat_mode') {
        // Mutually exclusive: 'none' | 'add' | 'included'
        line.vat_mode = val;
        // Cleanup legacy fields if present
        delete line.net_enabled;
        delete line.vat_enabled;
      } else {
        line[field] = val;
      }
      renderCreateDrawer();

      // Restore focus
      if (focusInfo) {
        const inputs = document.querySelectorAll('input');
        for (const inp of inputs) {
          if (inp.getAttribute('oninput') === focusInfo.oninput) {
            inp.focus();
            try { inp.setSelectionRange(focusInfo.selStart, focusInfo.selStart); } catch(e) {}
            break;
          }
        }
      }
    }

function onItemInput(e, lineId) {
      const val = e.target.value;
      const line = createWizard.data.lines.find(l => l.id === lineId);
      if (!line) return;
      line.item_code = val;
      // Auto-fill if exact match
      const match = mockItems.find(i => i.code === val);
      if (match) {
        applyItemToLine(line, match);
      } else {
        line.item_name = '';
        line.unit = '-';
      }
      showItemSuggestions(lineId, val);
    }

function applyItemToLine(line, item) {
      const du = itemDefaultUnit(item);
      line.item_name = item.name;
      line.unit = du ? du.uom : '-';
      line.unit_price = du ? du.price : line.unit_price;
      line.vat_mode = item.vat_group === 'VAT7' ? 'add' : 'none';
      line.vat_pct = item.vat_group === 'VAT7' ? 7 : line.vat_pct;
    }

function setLineUnit(lineId, uom) {
      const line = createWizard.data.lines.find(l => l.id === lineId);
      const item = mockItems.find(i => i.code === line?.item_code);
      if (!line || !item) return;
      line.unit = uom;
      line.unit_price = itemUnitPrice(item, uom);
      renderCreateDrawer();
    }

function onItemFocus(lineId) {
      itemSearchState.highlight = 0;
      const line = createWizard.data.lines.find(l => l.id === lineId);
      showItemSuggestions(lineId, line?.item_code || '');
    }

function showItemSuggestions(lineId, query) {
      document.querySelectorAll('.item-suggest-popover').forEach(n => n.remove());
      const input = document.getElementById(`item-input-${lineId}`);
      if (!input) return;
      const q = (query || '').toLowerCase().trim();
      const filtered = mockItems.filter(i => !q || i.code.toLowerCase().includes(q) || i.name.toLowerCase().includes(q)).slice(0, 6);
      itemSearchState.open = filtered.length > 0;
      itemSearchState.lineId = lineId;
      itemSearchState.filtered = filtered.map(i => i.code);
      if (itemSearchState.highlight >= filtered.length) itemSearchState.highlight = Math.max(0, filtered.length - 1);
      if (filtered.length === 0) return;
      const rect = input.getBoundingClientRect();
      const pop = document.createElement('div');
      pop.className = 'item-suggest-popover';
      pop.style.cssText = `position: fixed; left: ${rect.left}px; min-width: ${Math.max(360, rect.width)}px; background: white; border: 1px solid var(--c-border); border-radius: var(--r); box-shadow: var(--sh-lg); z-index: 70; max-height: 260px; overflow-y: auto;`;
      pop.innerHTML = filtered.map((i, idx) => {
        const du = itemDefaultUnit(i);
        const active = idx === itemSearchState.highlight;
        return `
        <button type="button" data-idx="${idx}" onmousedown="event.preventDefault()" onclick="pickItem('${lineId}','${i.code}')"
          style="width: 100%; text-align: left; padding: 8px 12px; background: ${active ? 'var(--c-primary-08)' : 'transparent'}; border: none; cursor: pointer; border-bottom: 1px solid var(--c-border-soft); display: flex; align-items: center; gap: 10px;">
          <i data-lucide="package" class="w-3.5 h-3.5" style="color: var(--c-text-soft); flex-shrink: 0;"></i>
          <div style="flex: 1; min-width: 0;">
            <div style="font-size: 12.5px; font-weight: 600; color: var(--c-navy);">${i.name}</div>
            <div class="tbl-mono" style="font-size: 11px; color: var(--c-text-mute);">${i.code} · ${(i.units||[]).length} หน่วย</div>
          </div>
          <span style="font-size: 11px; color: var(--c-text-soft); white-space: nowrap;">฿${du ? formatMoney(du.price) : '-'}/${du ? du.uom : ''}</span>
        </button>`;
      }).join('');
      // Rule #66 — Overlay Position Awareness: flip up when bottom space is short
      const spaceBelow = window.innerHeight - rect.bottom - 12;
      if (spaceBelow < 260 && rect.top - 12 > spaceBelow) {
        pop.style.bottom = (window.innerHeight - rect.top + 2) + 'px';
        pop.style.maxHeight = Math.min(260, rect.top - 12) + 'px';
      } else {
        pop.style.top = (rect.bottom + 2) + 'px';
        pop.style.maxHeight = Math.min(260, spaceBelow) + 'px';
      }
      document.body.appendChild(pop);
      renderIcons();
      const close = (ev) => { if (!pop.contains(ev.target) && ev.target !== input) { closeItemSuggest(); document.removeEventListener('click', close); } };
      setTimeout(() => document.addEventListener('click', close), 0);
    }

function closeItemSuggest() {
      document.querySelectorAll('.item-suggest-popover').forEach(n => n.remove());
      itemSearchState.open = false;
    }

function onItemKey(e, lineId) {
      if (!itemSearchState.open || itemSearchState.lineId !== lineId) {
        if (e.key === 'ArrowDown') { e.preventDefault(); showItemSuggestions(lineId, document.getElementById(`item-input-${lineId}`)?.value || ''); }
        return;
      }
      const n = (itemSearchState.filtered || []).length;
      if (e.key === 'ArrowDown') { e.preventDefault(); itemSearchState.highlight = Math.min(n - 1, itemSearchState.highlight + 1); showItemSuggestions(lineId, document.getElementById(`item-input-${lineId}`)?.value || ''); }
      else if (e.key === 'ArrowUp') { e.preventDefault(); itemSearchState.highlight = Math.max(0, itemSearchState.highlight - 1); showItemSuggestions(lineId, document.getElementById(`item-input-${lineId}`)?.value || ''); }
      else if (e.key === 'Enter') { e.preventDefault(); const code = (itemSearchState.filtered || [])[itemSearchState.highlight]; if (code) pickItem(lineId, code); }
      else if (e.key === 'Escape') { e.stopPropagation(); closeItemSuggest(); }  // Esc chain: close list first, drawer stays open
    }

function pickItem(lineId, code) {
      const item = mockItems.find(i => i.code === code);
      const line = createWizard.data.lines.find(l => l.id === lineId);
      if (item && line) {
        line.item_code = item.code;
        applyItemToLine(line, item);
      }
      document.querySelectorAll('.item-suggest-popover').forEach(n => n.remove());
      renderCreateDrawer();
    }
