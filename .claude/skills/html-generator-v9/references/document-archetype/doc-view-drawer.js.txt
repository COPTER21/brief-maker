function renderSignProgress(s) { const total = (chainOf(s).length || resolveDoa(s).steps.length) + 1; const done = signCount(s); const pct = Math.round(done / total * 100); const color = done === total ? function axisPill(v, labels) { const map = { none: { bg: 'rgba(115,117,123,0.10)', color: 'var(--c-text-mute)' }, partial: { bg: 'rgba(232,135,15,0.10)', color: 'var(--c-warning)' }, done: { bg: 'rgba(31,157,85,0.10)', color: 'var(--c-teal)' } }; const m = map[v] || map.none; return `<span style="display: inline-flex; align-items: center; gap: 3px; padding: 3px 8px; border-radius: 999px; font-size: 11px; font-weight: 600; background: ${m.bg}; color: ${m.color}; white-space: nowrap;"><span style="width: 4px; height: 4px; border-radius: 50%; background: ${m.color};"></span>${labels[v]}</span>`; }

function signCount(s) { const c = chainOf(s); return c.filter(x => x.status === 'approved').length + (s.submittedBy ? 1 : 0); }
