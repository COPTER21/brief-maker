  if (editId) { const s = findSO(editId); if (!s || sStatus(s) !== 'draft') { showToast('แก้ไขได้เฉพาะฉบับร่าง', 'warning'); navigate('view/' + editId); return; } createWizard.data = Object.assign(blankData(), { source_type: s.quotation ? 'qt' : 'new', quotation: s.quotation, customer: s.customer, shipTo: s.shipTo, channel: s.channel, rep: s.rep, warehouse: s.warehouse, orderDate: s.orderDate, deliveryDate: s.deliveryDate, paymentTerm: s.paymentTerm, custRef: s.custRef, note: s.note, coupon: s.coupon || '', couponAmt: s.couponAmt || 0, billTo: s.billTo || '', deliveryMethod: s.deliveryMethod || 'deliver', contact: s.contact || '', taxInvoice: s.taxInvoice !== false, campaign: s.campaign || '', promoDisabled: (s.promoDisabled || []).slice(), useTA: s.useTA !== false, lines: s.lines.map(l => Object.assign({}, l)), endbill: Object.assign({ enabled: false, mode: 'percent', value: 0, src: '' }, s.endbill || {}), promoApplied: (s.promoApplied || []).slice(), taRef: s.taRef, overrideNote: s.overrideNote, attachments: (s.attachments || []).slice() }); createWizard.step = 2; }
  else { createWizard.data = blankData(); if (from === 'qt') createWizard.data.source_type = 'qt'; if (from && from.startsWith('dup-')) { const src = findSO(from.slice(4)); if (src) { createWizard.data = Object.assign(blankData(), { customer: src.customer, shipTo: src.shipTo, channel: src.channel, rep: src.rep, warehouse: src.warehouse, paymentTerm: src.paymentTerm, note: src.note, lines: src.lines.filter(l => !l.is_free).map(l => Object.assign({}, l, { id: 'L' + Date.now() + Math.random().toString(36).slice(2, 5), override: false })) }); createWizard.step = 2; repriceAll(); } } }

  createWizard.step++; renderCreateDrawer(); }

function buildRecord() { const d = createWizard.data; return { customer: d.customer, shipTo: d.shipTo.trim(), channel: d.channel, rep: d.rep, warehouse: d.warehouse, orderDate: d.orderDate, deliveryDate: d.deliveryDate, paymentTerm: d.paymentTerm, custRef: d.custRef.trim(), quotation: d.quotation, note: d.note.trim(), coupon: (d.coupon || '').trim().toUpperCase(), couponAmt: d.couponAmt || 0, billTo: d.billTo.trim(), deliveryMethod: d.deliveryMethod, contact: d.contact.trim(), taxInvoice: !!d.taxInvoice, campaign: d.campaign, promoDisabled: d.promoDisabled.slice(), useTA: !!d.useTA, lines: d.lines.filter(l => l.is_free || l.item_code).map(l => Object.assign({}, l)), endbill: Object.assign({}, d.endbill), promoApplied: d.promoApplied.slice(), taRef: d.taRef, priceOverride: d.lines.some(l => l.override), overrideNote: d.overrideNote.trim(), attachments: d.attachments.slice() }; }

function persist() { const data = buildRecord(); let s; if (createWizard.editId) { s = findSO(createWizard.editId); Object.assign(s, data); pushAudit(s, 'แก้ไขร่างคำสั่งขาย'); } else { s = seedAudit(mkSO(Object.assign({ code: nextCode(), status: 'draft', createdBy: ME.name, createdAt: nowIso(), updatedBy: ME.name, updatedAt: nowIso() }, data))); if (data.quotation) s.audit[0].act = 'สร้างคำสั่งขายจากใบเสนอราคา ' + data.quotation; SOS.unshift(s); } return s; }

function saveDraft() { const d = createWizard.data; if (d.lines.some(l => l.override) && CFG.priceOverrideNeedsNote && !d.overrideNote.trim()) { markErr('f-overrideNote', true); return showToast('ระบุเหตุผลการแก้ราคาก่อนบันทึก', 'warning'); } const s = persist(); showToast(`บันทึกแบบร่าง ${s.code} เรียบร้อย`, 'success'); document.getElementById('overlay-root').innerHTML = ''; navigate('view/' + s.id); }

function submitForApproval() { const d = createWizard.data; if (createWizard._over) return showToast('เกินวงเงินเครดิต — ส่งอนุมัติไม่ได้ (บันทึกร่างได้)', 'warning'); if (d.lines.some(l => l.override) && CFG.priceOverrideNeedsNote && !d.overrideNote.trim()) { markErr('f-overrideNote', true); return showToast('ระบุเหตุผลการแก้ราคาก่อนส่งอนุมัติ', 'warning'); } const s = persist(); document.getElementById('overlay-root').innerHTML = ''; navigate('view/' + s.id); setTimeout(() => openSubmitModal(s.id), 250); }

function combo(key, display, ph) { return `<div style="position: relative;"><i data-lucide="search" class="w-3.5 h-3.5" style="position: absolute; left: 10px; top: 50%; transform: translateY(-50%); color: var(--c-text-soft);"></i><input type="text" id="cbi-${key}" autocomplete="off" value="${display}" placeholder="${ph}" class="input" style="padding-left: 30px;" onfocus="comboOpen('${key}', '')" oninput="comboOpen('${key}', this.value)" onkeydown="comboKey(event, '${key}')"></div>`; }

function comboItems(key, q) { q = (q || '').toLowerCase().trim(); if (key === 'cust') return CUSTOMERS.filter(c => !q || (c.name + ' ' + c.code).toLowerCase().includes(q)).map(c => ({ val: c.code, t: c.name, s: `${c.code} · ${(GROUP_BY[c.group] || {}).name} · ${c.credit ? 'เครดิตคงเหลือ ฿' + formatMoney(creditAvail(c)) : 'เงินสด'}`, ico: 'building-2' })); if (key === 'rep') return EMPLOYEES.filter(e => !q || (e.name + ' ' + e.dept).toLowerCase().includes(q)).map(e => ({ val: e.name, t: e.name, s: `${ROLE_LABEL[e.role] || e.role} · ${e.dept}`, ico: 'user' })); if (key.startsWith('slot-')) { const st = modalState.slots[+key.slice(5)]; return EMPLOYEES.filter(e => st.roles.includes(e.role) && e.name !== ME.name && (!q || (e.name + ' ' + e.dept + ' ' + e.code).toLowerCase().includes(q))).map(e => ({ val: e.name, t: e.name, s: `${ROLE_LABEL[e.role] || e.role} · ${e.dept} · ${e.code}`, av: e.initials })); } return []; }

function comboPick(key, val) { if (key === 'cust') { setCustomer(val); renderCreateDrawer(); } else if (key === 'rep') { createWizard.data.rep = val; renderCreateDrawer(); } else if (key.startsWith('slot-')) { modalState.slots[+key.slice(5)].assignee = val; renderSubmitModal(); } document.querySelectorAll('.combo-pop').forEach(n => n.remove()); comboState.key = null; }

function comboOpen(key, q) { document.querySelectorAll('.combo-pop').forEach(n => n.remove()); const input = document.getElementById('cbi-' + key); if (!input) return; comboState.key = key; comboState.list = comboItems(key, q); comboState.hi = 0; const rect = input.getBoundingClientRect(); const pop = document.createElement('div'); pop.className = 'combo-pop'; pop.id = 'combo-pop'; const up = window.innerHeight - rect.bottom < 280; pop.style.cssText = `left: ${rect.left}px; width: ${Math.max(rect.width, 320)}px; ${up ? 'bottom: ' + (window.innerHeight - rect.top + 2) + 'px;' : 'top: ' + (rect.bottom + 2) + 'px;'}`; document.body.appendChild(pop); paintCombo(); const close = ev => { if (!pop.contains(ev.target) && ev.target !== input) { pop.remove(); document.removeEventListener('click', close); } }; setTimeout(() => document.addEventListener('click', close), 0); }

function paintCombo() { const pop = document.getElementById('combo-pop'); if (!pop) return; pop.innerHTML = comboState.list.length ? comboState.list.map((it, k) => `<button type="button" class="${k === comboState.hi ? 'hi' : ''}" onmousedown="event.preventDefault()" onclick="comboPick('${comboState.key}','${it.val}')">${it.av ? `<span class="emp-av" style="width:26px;height:26px;font-size:10px;">${it.av}</span>` : `<i data-lucide="${it.ico}" class="w-3.5 h-3.5" style="color: var(--c-text-soft); flex-shrink: 0;"></i>`}<div style="flex: 1; min-width: 0;"><div style="font-size: 12.5px; font-weight: 600; color: var(--c-navy);">${it.t}</div><div style="font-size: 11px; color: var(--c-text-mute);">${it.s}</div></div></button>`).join('') : `<div style="padding: 10px 12px; font-size: 12px; color: var(--c-text-soft);">ไม่พบรายการ</div>`; renderIcons(); }

function comboKey(e, key) { if (comboState.key !== key || !document.getElementById('combo-pop')) return; if (e.key === 'ArrowDown') { comboState.hi = Math.min(comboState.hi + 1, comboState.list.length - 1); paintCombo(); e.preventDefault(); } else if (e.key === 'ArrowUp') { comboState.hi = Math.max(comboState.hi - 1, 0); paintCombo(); e.preventDefault(); } else if (e.key === 'Enter') { const it = comboState.list[comboState.hi]; if (it) comboPick(key, it.val); e.preventDefault(); } else if (e.key === 'Escape') { document.querySelectorAll('.combo-pop').forEach(n => n.remove()); e.stopPropagation(); } }
