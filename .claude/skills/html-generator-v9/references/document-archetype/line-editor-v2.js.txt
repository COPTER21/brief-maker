function createBlankLine() { return { id: 'L' + Date.now() + Math.random().toString(36).slice(2, 5), item_code: '', item_name: '', qty: 1, unit: '-', unit_price: 0, discount_pct: 0, vat_mode: 'add', vat_pct: CFG.vatPct, note: '', price_src: '', ta_src: '', promo_disc: 0, is_free: false, override: false }; }

function migrateLine(l) { if (l.vat_mode == null) { l.vat_mode = l.net_enabled ? 'included' : l.vat_enabled === false ? 'none' : 'add'; } if (l.vat_pct == null) l.vat_pct = CFG.vatPct; return l; }

function calcLineVat(l) { migrateLine(l); const subtotal = (+l.qty || 0) * (+l.unit_price || 0); const discount = (l.discount_mode || 'percent') === 'amount' ? Math.min(subtotal, +l.discount_amt || 0) : subtotal * ((+l.discount_pct || 0) / 100); const netAmount = subtotal - discount; let vatAmount = 0; let lineTotal = netAmount; if (l.vat_mode === 'add') { vatAmount = netAmount * (l.vat_pct / 100); lineTotal = netAmount + vatAmount; } else if (l.vat_mode === 'included') { vatAmount = netAmount * (l.vat_pct / 100) / (1 + l.vat_pct / 100); lineTotal = netAmount; } return { netAmount, vatAmount, lineTotal }; }

function totals(so) { let before = 0, vat = 0; so.lines.forEach(l => { migrateLine(l); const { netAmount, vatAmount } = calcLineVat(l); before += l.vat_mode === 'included' ? netAmount - vatAmount : netAmount; vat += vatAmount; }); const after = before + vat; const eb = so.endbill || {}; let ebAmt = 0; if (eb.enabled) ebAmt = eb.mode === 'amount' ? (+eb.value || 0) : after * ((+eb.value || 0) / 100); const cp = +so.couponAmt || 0; return { before, vat, after, ebAmt, coupon: cp, grand: Math.max(0, after - ebAmt - cp), n: so.lines.length }; }

function taxBadgeV(l) { migrateLine(l); if (l.vat_mode === 'none') return `<span style="display: inline-flex; align-items: center; gap: 3px; font-size: 11px; padding: 3px 8px; border-radius: 999px; background: rgba(115,117,123,0.10); color: var(--c-text-mute); font-weight: 600; white-space: nowrap;"><span style="width: 4px; height: 4px; border-radius: 50%; background: var(--c-text-mute);"></span>ไม่คิด</span>`; if (l.vat_mode === 'included') return `<span style="display: inline-flex; align-items: center; gap: 3px; font-size: 11px; padding: 3px 8px; border-radius: 999px; background: rgba(255,59,48,0.10); color: var(--c-primary); font-weight: 600; white-space: nowrap;" title="ราคารวม VAT · คำนวณย้อน"><span style="width: 4px; height: 4px; border-radius: 50%; background: var(--c-primary);"></span>NET ${l.vat_pct}%</span>`; return `<span style="display: inline-flex; align-items: center; gap: 3px; font-size: 11px; padding: 3px 8px; border-radius: 999px; background: rgba(255,154,31,0.10); color: var(--c-purple); font-weight: 600; white-space: nowrap;"><span style="width: 4px; height: 4px; border-radius: 50%; background: var(--c-purple);"></span>VAT ${l.vat_pct}%</span>`; }

  if (field === 'qty') { line.qty = Math.max(1, parseInt(val) || 1); repriceAll(); } else if (field === 'unit_price') { const v = Math.max(0, parseFloat(val) || 0); line.unit_price = v; line.override = Math.abs(v - (line._sysPrice || 0)) > 0.005; } else if (field === 'discount_pct') { line.discount_pct = Math.min(100, Math.max(0, parseFloat(val) || 0)); } else if (field === 'vat_pct') { line.vat_pct = parseFloat(val) || 0; } else if (field === 'vat_mode') { line.vat_mode = val; delete line.net_enabled; delete line.vat_enabled; } else line[field] = val;

function onItemKey(e, lineId) { if (!itemSearchState.open || itemSearchState.lineId !== lineId) return; if (e.key === 'ArrowDown') { itemSearchState.highlight = Math.min(itemSearchState.highlight + 1, itemSearchState.list.length - 1); paintItemPop(); e.preventDefault(); } else if (e.key === 'ArrowUp') { itemSearchState.highlight = Math.max(itemSearchState.highlight - 1, 0); paintItemPop(); e.preventDefault(); } else if (e.key === 'Enter') { const it = itemSearchState.list[itemSearchState.highlight]; if (it) pickItem(lineId, it.code); e.preventDefault(); } else if (e.key === 'Escape') { document.querySelectorAll('.combo-pop').forEach(n => n.remove()); itemSearchState.open = false; e.stopPropagation(); } }
