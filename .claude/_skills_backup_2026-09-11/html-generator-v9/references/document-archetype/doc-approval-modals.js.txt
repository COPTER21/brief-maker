function modalShell(inner) { const m = document.createElement('div'); m.className = 'modal-overlay'; m.innerHTML = `<div class="backdrop" onclick="this.parentElement.remove();"></div>${inner}`; document.body.appendChild(m); renderIcons(); return m; }

function closeModal() { document.querySelectorAll('.modal-overlay').forEach(n => n.remove()); }

function openSubmitModal(id) { const s = findSO(id); const short = atpCheck(s); if (short.length) return showToast('ส่งอนุมัติไม่ได้ — สต๊อกไม่พอ ' + short.length + ' รายการ (' + short.map(x => x.p.name + ' ขาด ' + fmi(x.need - x.atp)).join(', ') + ') · แก้ไขร่างก่อน', 'error'); const res = resolveDoa(s); modalState.id = id; modalState.slots = res.steps.map(st => ({ roles: st.roles, assignee: null })); modalState.err = false; closeModal(); modalShell(`<div class="card" id="submit-modal" style="position: relative; max-width: 780px; width: 100%; box-shadow: var(--sh-xl);"></div>`); renderSubmitModal(); }

function confirmSubmit() { const s = findSO(modalState.id); const sl = modalState.slots; if (sl.some(st => !st.assignee)) { modalState.err = true; renderSubmitModal(); return showToast('เลือกผู้อนุมัติให้ครบทุกขั้น', 'warning'); } const res = resolveDoa(s); s.status = 'pending_approval'; s.approval_status = 'pending_approval'; s.submittedBy = ME.name; s.submittedAt = nowIso(); s.doa_entry_ref = res.doa_entry_ref; s.approval_chain = res.steps.map((st, i) => ({ roles: st.roles, assignee: sl[i].assignee, status: 'pending', by: null, at: null })); s.approver_role = s.approval_chain[0].roles[0]; s.approved_by = null; s.approved_at = null; pushAudit(s, 'ส่งอนุมัติ · ' + s.approval_chain.map((st, i) => (i + 1) + ') ' + st.assignee + ' (' + (ROLE_LABEL[st.roles[0]] || st.roles[0]) + ')').join(' → '), res.why.join(' · ')); closeModal(); showToast(`ส่ง ${s.code} เพื่ออนุมัติแล้ว — My Approval ของ ${s.approval_chain[0].assignee}`, 'success'); if (getRoute().name === 'view') renderViewDrawer(); else render(); }

'ขั้นสุดท้าย — ยืนยันคำสั่งขาย · ส่ง Warehouse จัดของ · นับเครดิต · แจ้งผู้ส่ง (doa_result)' : 'ยังมีขั้นถัดไป — ส่งต่อผ่าน My Approval'}</div></div></div>

function confirmApprove(id) { const s = findSO(id); const cur = currentStep(s); if (!canActStep(s)) return showToast('ไม่ใช่ผู้มีสิทธิ์ในขั้นปัจจุบัน', 'warning'); const note = (document.getElementById('md-note') || {}).value || ''; cur.status = 'approved'; cur.by = ME.name; cur.at = nowIso(); cur.note = note; const idx = chainOf(s).indexOf(cur); const next = currentStep(s);
  if (next) { s.approver_role = next.roles[0]; pushAudit(s, `อนุมัติขั้น ${idx + 1}/${chainOf(s).length} · ${ME.name} (${ROLE_LABEL[cur.roles[0]] || cur.roles[0]})`, note); showToast(`อนุมัติขั้น ${idx + 1} แล้ว — ส่งต่อ ${next.assignee}`, 'success'); } else { s.status = 'confirmed'; s.approval_status = 'approved'; s.approved_by = ME.name; s.approved_at = nowIso(); s.approver_role = null; s.confirmedBy = ME.name; s.confirmedAt = nowIso(); const c = CUST_BY[s.customer]; if (c) c.openSO += creditExposure(s); const rs = reserveSO(s); const back = rs.filter(x => x.back > 0); pushAudit(s, `อนุมัติขั้น ${idx + 1}/${chainOf(s).length} · ${ME.name} (${ROLE_LABEL[cur.roles[0]] || cur.roles[0]}) — ครบสาย ยืนยันคำสั่งขาย · จองสต๊อก ${rs.length} รายการ${back.length ? ' (backorder ' + back.length + ')' : ''}${CFG.linkage.delivery && s.deliveryMethod !== 'pickup' ? ' · ส่ง Warehouse ออก DN' : ''}${s.taRef ? ' · นับเข้ายอดผูกพัน ' + s.taRef : ''}`, note); showToast(`อนุมัติครบสาย — ยืนยัน · จองสต๊อก ${rs.length} รายการ${back.length ? ' · backorder ' + back.length : ''} · นับเครดิต · ENG-NOTIFY so.confirmed`, 'success'); }
  closeModal(); if (getRoute().name === 'view') renderViewDrawer(); else render(); }

function reasonModal(title, icon, iconColor, iconBg, desc, extraFields, submitLabel, submitCls, onOk) { closeModal(); modalShell(`<div class="card" style="position: relative; max-width: 520px; width: 100%; box-shadow: var(--sh-xl);"><div style="padding: 18px; border-bottom: 1px solid var(--c-border-soft); display: flex; align-items: flex-start; gap: 12px;"><div style="width: 40px; height: 40px; border-radius: 50%; background: ${iconBg}; display: grid; place-items: center; flex-shrink: 0;"><i data-lucide="${icon}" class="w-5 h-5" style="color: ${iconColor};"></i></div><div><h3 style="font-size: 14px; font-weight: 700; color: var(--c-navy);">${title}</h3><p style="font-size: 13px; margin-top: 4px; color: var(--c-text-mute);">${desc}</p></div></div><div style="padding: 18px;">${extraFields || ''}<label style="display: block; font-size: 13px; font-weight: 500; margin-bottom: 6px; color: var(--c-navy);">เหตุผล <span style="color: var(--c-danger);">*</span></label><textarea id="md-reason" rows="3" placeholder="ระบุเหตุผล..." class="input" style="height: auto; padding: 10px 12px; resize: vertical;"></textarea></div><div style="padding: 12px 18px; background: var(--c-off-white); border-top: 1px solid var(--c-border-soft); display: flex; align-items: center; justify-content: flex-end; gap: 8px; border-radius: 0 0 var(--r-lg) var(--r-lg);"><button onclick="closeModal()" class="btn btn-ghost">ปิด</button><button id="md-ok" class="btn ${submitCls}"><i data-lucide="${icon}" class="w-4 h-4"></i><span>${submitLabel}</span></button></div></div>`); document.getElementById('md-ok').onclick = () => { const r = (document.getElementById('md-reason').value || '').trim(); if (!r) return showToast('กรุณาระบุเหตุผล', 'warning'); onOk(r); }; }

function openRejectModal(id) { const s = findSO(id); reasonModal(`ไม่อนุมัติ ${s.code} — ตีกลับเป็นร่าง`, 'undo-2', 'var(--c-danger)', 'var(--c-danger-08)', 'เอกสารจะกลับเป็นฉบับร่างให้ผู้ส่งแก้ไขแล้วส่งใหม่ · ประวัติเก็บ append-only', '', 'ไม่อนุมัติ', 'btn-danger', r => { const cur = currentStep(s); if (!canActStep(s)) return showToast('ไม่ใช่ผู้มีสิทธิ์ในขั้นปัจจุบัน', 'warning'); cur.status = 'rejected'; cur.by = ME.name; cur.at = nowIso(); cur.note = r; s.approval_history = (s.approval_history || []).concat([{ submittedBy: s.submittedBy, submittedAt: s.submittedAt, chain: s.approval_chain, result: 'rejected' }]); s.status = 'draft'; s.approval_status = 'draft'; s.approved_by = null; s.approver_role = null; s.approval_chain = null; pushAudit(s, 'ไม่อนุมัติ · ' + ME.name + ' — ตีกลับเป็นร่าง', r); closeModal(); showToast(`ตีกลับ ${s.code} เพื่อแก้ไข`, 'info'); renderViewDrawer(); }); }

function openCancelModal(id) { const s = findSO(id); const st = sStatus(s); const hasDN = s.docs.dn.length > 0; if (hasDN) return showToast('มีการจัดส่งแล้ว — ยกเลิกไม่ได้ ให้ทำ Sales Return (RMA) / ใบลดหนี้', 'warning'); reasonModal(`ยกเลิกคำสั่งขาย ${s.code}?`, 'x-circle', 'var(--c-danger)', 'var(--c-danger-08)', 'การยกเลิกจะไม่สามารถย้อนกลับได้ · เครดิตที่กันไว้ปล่อยคืน · ถ้าส่ง Warehouse แล้วจะแจ้งยกเลิกการจัด', `<label style="display: block; font-size: 13px; font-weight: 500; margin-bottom: 6px; color: var(--c-navy);">ประเภทเหตุผล</label><select id="md-rsel" class="select" style="margin-bottom: 12px;">${REASONS.cancel.map(x => `<option>${x}</option>`).join('')}</select>`, 'ยืนยันยกเลิก', 'btn-danger', r => { const rs = (document.getElementById('md-rsel') || {}).value || ''; chainOf(s).forEach(x => { if (x.status === 'pending') x.status = 'cancelled'; }); const wasOpen = ['confirmed', 'in_progress', 'on_hold'].includes(st); s.status = 'cancelled'; s.approval_status = 'cancelled'; s.hold = null; s.cancel = { by: ME.name, at: nowIso(), reason: rs, note: r }; if (wasOpen) { const c = CUST_BY[s.customer]; if (c) c.openSO = Math.max(0, c.openSO - creditExposure(s)); releaseSO(s); } pushAudit(s, 'ยกเลิกคำสั่งขาย' + (wasOpen ? ' · ปล่อยสต๊อกที่จอง · ปล่อยเครดิต' : ''), rs + ' — ' + r); closeModal(); showToast(`ยกเลิก ${s.code} เรียบร้อย${wasOpen ? ' · ปล่อยเครดิต · แจ้ง Warehouse' : ''}`, 'success'); if (getRoute().name === 'view') renderViewDrawer(); else renderTableOnly(); }); }

function openHoldModal(id) { const s = findSO(id); reasonModal(`พักคำสั่งขาย ${s.code} (hold)`, 'pause', 'var(--c-warning)', 'var(--c-warning-08)', 'Warehouse หยุดจัดส่งจนกว่าจะปลดพัก · เครดิตยังกันไว้', `<label style="display: block; font-size: 13px; font-weight: 500; margin-bottom: 6px; color: var(--c-navy);">ประเภทเหตุผล</label><select id="md-rsel" class="select" style="margin-bottom: 12px;">${REASONS.hold.map(x => `<option>${x}</option>`).join('')}</select>`, 'พัก', 'btn-primary', r => { const rs = (document.getElementById('md-rsel') || {}).value || ''; s.status = 'on_hold'; s.hold = { by: ME.name, at: nowIso(), reason: rs, note: r }; pushAudit(s, 'พักคำสั่งขาย (hold)', rs + ' — ' + r); closeModal(); showToast(`พัก ${s.code} แล้ว · แจ้ง Warehouse หยุดจัด`, 'success'); renderViewDrawer(); }); }

function openManualModal(id, k) { const s = findSO(id); const lab = { delivery: 'จัดส่ง', invoice: 'วางบิล', payment: 'ชำระเงิน' }[k]; const L = { delivery: DLV, invoice: INVL, payment: PAY }[k]; reasonModal(`อัพเดตสถานะ${lab} (module ไม่เชื่อม)`, 'sliders-vertical', 'var(--c-primary)', 'var(--c-primary-08)', `${s.code} — Sales Configuration ตั้งไม่เชื่อม ${k} · บันทึกสถานะด้วยมือ (audit)`, `<label style="display: block; font-size: 13px; font-weight: 500; margin-bottom: 6px; color: var(--c-navy);">สถานะ</label><select id="md-lsel" class="select" style="margin-bottom: 12px;">${['none', 'partial', 'done'].map(v => `<option value="${v}" ${effLink(s, k) === v ? 'selected' : ''}>${L[v]}</option>`).join('')}</select>`, 'บันทึก', 'btn-primary', r => { const v = (document.getElementById('md-lsel') || {}).value || 'none'; s.manual[k] = v; pushAudit(s, `อัพเดตสถานะ${lab}ด้วยมือ → ${L[v]}`, r); closeModal(); showToast('บันทึกสถานะแล้ว', 'success'); viewState.tab = 'fin'; renderViewDrawer(); }); }

function resolveDoa(s) { const set = MOCK_DOA_ENTRY.matrix[0]; return { doa_entry_ref: MOCK_DOA_ENTRY.doa_entry_ref, chainMode: MOCK_DOA_ENTRY.chainMode, steps: set.steps.map(st => ({ roles: st.roles.slice() })), why: ['ไม่มีวงเงิน — สายเดียวทุกใบ อนุมัติตามลำดับตำแหน่ง'] }; }


function chainOf(r) { return Array.isArray(r.approval_chain) ? r.approval_chain : []; }

function currentStep(r) { const c = chainOf(r); return c.find(st => st.status === 'pending') || null; }

function canActStep(r) { const st = currentStep(r); if (!st || r.submittedBy === ME.name) return false; return st.assignee ? st.assignee === ME.name : st.roles.includes(ME.role); }

function chainLabel(r) { const c = chainOf(r); return c.length ? c.map(st => st.roles.map(x => ROLE_LABEL[x] || x).join('/')).join(' → ') : '—'; }
